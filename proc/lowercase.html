pure function lowercase(s) result(l)
    !! `s` in lowercase (see [[lower_char]]).
    character(len=*), intent(in) :: s   !! the text
    character(len=len(s)) :: l
    integer :: k
    do k = 1, len(s)
        l(k:k) = lower_char(s(k:k))
    end do
end function lowercase
